# 資料庫操作
def get_user(user_id: int):
    c.execute('SELECT draws, last_login FROM users WHERE user_id=?', (user_id,))
    return c.fetchone()

def create_user(user_id: int, draws: int):
    today = datetime.date.today().isoformat()
    c.execute('INSERT INTO users (user_id, draws, last_login) VALUES (?, ?, ?)', (user_id, draws, today))
    conn.commit()

def update_user_draws(user_id: int, draws: int):
    c.execute('UPDATE users SET draws=? WHERE user_id=?', (draws, user_id))
    conn.commit()

def update_last_login(user_id: int, date_str: str):
    c.execute('UPDATE users SET last_login=? WHERE user_id=?', (date_str, user_id))
    conn.commit()

def add_draws(user_id: int, amount: int):
    row = get_user(user_id)
    if row:
        update_user_draws(user_id, row[0] + amount)
    else:
        create_user(user_id, amount)

# 每日送 3 抽
async def daily_reset():
    await bot.wait_until_ready()
    while not bot.is_closed():
        today = datetime.date.today().isoformat()
        c.execute('SELECT user_id, draws, last_login FROM users')
        for user_id, draws, last_login in c.fetchall():
            if last_login != today:
                c.execute('UPDATE users SET draws=?, last_login=? WHERE user_id=?', (draws + 3, today, user_id))
        conn.commit()
        tomorrow = datetime.datetime.combine(datetime.date.today() + datetime.timedelta(days=1), datetime.time(0, 1))
        await asyncio.sleep((tomorrow - datetime.datetime.now()).total_seconds())

# 登錄指令
@tree.command(name="login", description="登錄你的抽數")
@app_commands.describe(amount="要新增的抽數")
async def login(interaction: discord.Interaction, amount: int):
    if amount <= 0:
        await interaction.response.send_message("請輸入正整數抽數。", ephemeral=True)
        return
    user_id = interaction.user.id
    if not get_user(user_id):
        create_user(user_id, amount)
        await interaction.response.send_message(f"首次登錄成功，已新增 {amount} 抽，每日自動送 3 抽！")
    else:
        add_draws(user_id, amount)
        await interaction.response.send_message(f"已新增 {amount} 抽，目前剩餘 {get_user(user_id)[0]} 抽。")

# 抽卡指令
@tree.command(name="gatch", description="從卡池抽卡")
@app_commands.describe(pool="卡池名稱", num="抽卡次數")
async def gatch(interaction: discord.Interaction, pool: str, num: int):
    pool = pool.lower()
    if pool not in pools:
        await interaction.response.send_message(f"卡池不存在，目前支援：{', '.join(pools.keys())}", ephemeral=True)
        return
    if num <= 0:
        await interaction.response.send_message("請輸入大於 0 的抽卡數。", ephemeral=True)
        return

    user_id = interaction.user.id
    row = get_user(user_id)
    if not row:
        await interaction.response.send_message("請先使用 /login 登錄。", ephemeral=True)
        return
    draws = row[0]
    if draws < num:
        await interaction.response.send_message(f"抽數不足，剩餘 {draws} 抽。", ephemeral=True)
        return

    update_user_draws(user_id, draws - num)

    results = []
    for _ in range(num):
        if random.random() < 0.01:
            results.append(f"🎉 {random.choice(hidden_items)}")
        else:
            results.append(random.choice(pools[pool]))

    await interaction.response.send_message(
        f"你抽到了：\n" + "\n".join(results) + f"\n\n剩餘抽數：{draws - num}"
    )

# 查看抽數
@tree.command(name="check", description="查看你的剩餘抽數")
async def check(interaction: discord.Interaction):
    row = get_user(interaction.user.id)
    if row:
        await interaction.response.send_message(f"你目前有 {row[0]} 抽。")
    else:
        await interaction.response.send_message("你還沒有登錄，請先使用 /login。")

# 上線同步
@bot.event
async def on_ready():
    print(f"✅ Bot 上線：{bot.user}")
    await tree.sync()
    bot.loop.create_task(daily_reset())
