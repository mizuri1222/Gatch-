# SQLite 初始化
conn = sqlite3.connect('gacha.db')
c = conn.cursor()
c.execute('''
CREATE TABLE IF NOT EXISTS users (
    user_id INTEGER PRIMARY KEY,
    draws INTEGER NOT NULL,
    last_login TEXT
)
''')
conn.commit()

# Discord Bot 初始化
intents = discord.Intents.default()
bot = discord.Client(intents=intents)
tree = app_commands.CommandTree(bot)


# 資料庫操作
def get_user(user_id: int):
    c.execute('SELECT draws, last_login FROM users WHERE user_id=?', (user_id,))
    return c.fetchone()


def create_user(user_id: int, draws: int):
    c.execute('INSERT INTO users (user_id, draws, last_login) VALUES (?, ?, ?)', (user_id, draws, datetime.date.today().isoformat()))
    conn.commit()


def update_user_draws(user_id: int, draws: int):
    c.execute('UPDATE users SET draws=? WHERE user_id=?', (draws, user_id))
    conn.commit()


def update_last_login(user_id: int, date_str: str):
    c.execute('UPDATE users SET last_login=? WHERE user_id=?', (date_str, user_id))
    conn.commit()


def add_draws(user_id: int, amount: int):
    row = get_user(user_id)
    if row:
        update_user_draws(user_id, row[0] + amount)
    else:
        create_user(user_id, amount)


# 每日送3抽
async def daily_reset():
    await bot.wait_until_ready()
    while not bot.is_closed():
        today = datetime.date.today()
        c.execute('SELECT user_id, draws, last_login FROM users')
        for user_id, draws, last_login in c.fetchall():
            if last_login != today.isoformat():
                c.execute('UPDATE users SET draws=?, last_login=? WHERE user_id=?',
                          (draws + 3, today.isoformat(), user_id))
        conn.commit()
        # 等待到明天00:01再執行
        now = datetime.datetime.now()
        tomorrow = datetime.datetime.combine(now.date() + datetime.timedelta(days=1), datetime.time(0, 1))
        await asyncio.sleep((tomorrow - now).total_seconds())


# 登錄抽數指令
@tree.command(name="login", description="登錄你的抽數")
@app_commands.describe(amount="新增的抽數")
async def login(interaction: discord.Interaction, amount: int):
    if amount <= 0:
        await interaction.response.send_message("請輸入正整數的抽數。", ephemeral=True)
        return
    user_id = interaction.user.id
    row = get_user(user_id)
    if row is None:
        create_user(user_id, amount)
        await interaction.response.send_message(f"首次登錄成功，已新增 {amount} 抽。以後每天會自動送3抽！")
    else:
        add_draws(user_id, amount)
        draws = get_user(user_id)[0]
        await interaction.response.send_message(f"成功新增 {amount} 抽，目前剩餘 {draws} 抽。")

# 抽卡指令
@tree.command(name="gatch", description="從 basic 卡池抽卡（含隱藏版角色）")
@app_commands.describe(num="抽卡次數")
async def gatch(interaction: discord.Interaction, num: int):
    if num <= 0:
        await interaction.response.send_message("抽卡次數必須大於 0。", ephemeral=True)
        return
    user_id = interaction.user.id
    row = get_user(user_id)
    if row is None:
        await interaction.response.send_message("你還沒有登錄抽數，請先使用 /login。", ephemeral=True)
        return
    if row[0] < num:
        await interaction.response.send_message(f"抽數不足，目前剩餘 {row[0]} 抽。", ephemeral=True)
        return


    update_user_draws(user_id, row[0] - num)


    results = []
    for _ in range(num):
        if random.random() < 0.01:
            pick = random.choice(hidden_pool)
        else:
            pick = random.choice(normal_pool)
        results.append(pick)


    await interaction.response.send_message(f"你抽到了：\n" + "\n".join(results) + f"\n\n剩餘抽數：{row[0] - num}")


# 啟動事件
@bot.event
async def on_ready():
    print(f"✅ Bot 上線：{bot.user}")
    await tree.sync()
    bot.loop.create_task(daily_reset())
